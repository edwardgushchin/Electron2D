cmake_minimum_required(VERSION 3.24)
project(Electron2DCrossNative LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(EMSCRIPTEN)
    add_compile_options(-fwasm-exceptions -sSUPPORT_LONGJMP=wasm)
    add_link_options(-fwasm-exceptions -sSUPPORT_LONGJMP=wasm)
endif()
set(ELECTRON2D_STATIC_NATIVE OFF)
if(APPLE OR EMSCRIPTEN)
    set(ELECTRON2D_STATIC_NATIVE ON)
endif()
set(BUILD_SDL3 ON CACHE BOOL "" FORCE)
set(XNASONG OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_UTILS OFF CACHE BOOL "" FORCE)
set(base "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_subdirectory("${base}/font-native" font)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
if(ELECTRON2D_STATIC_NATIVE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
endif()
if(EMSCRIPTEN)
    FetchContent_Declare(SDL3
        URL https://www.libsdl.org/release/SDL3-3.4.16.tar.gz
        URL_HASH SHA256=7322236cd12090c3eb40b9728be4d49c76f66ad17d04369584d4ecad5cf77c68)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(SDL3)
    set(SDL3_LIBRARIES SDL3::SDL3-static)
endif()
add_subdirectory("${base}/audio-native" audio)
add_subdirectory("${base}/../src/Servers/Text/Native" text)
add_subdirectory("${base}/enet-native" enet)
if(EMSCRIPTEN)
    add_library(Electron2DWasmCompat STATIC wasm_sjlj.c)
    target_link_libraries(Electron2DFreeTypeCheck PRIVATE Electron2DWasmCompat)
    add_executable(Electron2DWasmJumpCheck EXCLUDE_FROM_ALL test_wasm_sjlj.c)
    target_link_libraries(Electron2DWasmJumpCheck PRIVATE Electron2DWasmCompat)
    target_link_options(Electron2DWasmJumpCheck PRIVATE -sWASM_ASYNC_COMPILATION=0)
endif()
