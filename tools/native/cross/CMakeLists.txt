cmake_minimum_required(VERSION 3.24)
project(Electron2DCrossNative LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(EMSCRIPTEN)
    add_compile_options(-fwasm-exceptions -sSUPPORT_LONGJMP=wasm)
    add_link_options(-fwasm-exceptions -sSUPPORT_LONGJMP=wasm)
endif()
set(ELECTRON2D_STATIC_NATIVE OFF)
if(APPLE OR EMSCRIPTEN)
    set(ELECTRON2D_STATIC_NATIVE ON)
endif()
set(BUILD_SDL3 ON CACHE BOOL "" FORCE)
set(XNASONG OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_UTILS OFF CACHE BOOL "" FORCE)
set(base "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_subdirectory("${base}/font-native" font)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
if(ELECTRON2D_STATIC_NATIVE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
endif()
if(EMSCRIPTEN)
    FetchContent_Declare(SDL3
        URL https://www.libsdl.org/release/SDL3-3.4.18.tar.gz
        URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(SDL3)
    set(SDL3_LIBRARIES SDL3::SDL3-static)
    FetchContent_Declare(SDLImage
        GIT_REPOSITORY https://github.com/libsdl-org/SDL_image.git
        GIT_TAG f661fa1ad24ab1b81e43662532f9a6a9fcf67ea6
        GIT_SUBMODULES external/libwebp)
    set(SDLIMAGE_VENDORED ON CACHE BOOL "" FORCE)
    set(SDLIMAGE_STRICT ON CACHE BOOL "" FORCE)
    set(SDLIMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
    set(SDLIMAGE_TESTS OFF CACHE BOOL "" FORCE)
    set(SDLIMAGE_INSTALL OFF CACHE BOOL "" FORCE)
    foreach(format ANI AVIF GIF JXL LBM PCX PNG PNM QOI TIF XCF XPM XV)
        set(SDLIMAGE_${format} OFF CACHE BOOL "" FORCE)
    endforeach()
    FetchContent_MakeAvailable(SDLImage)
endif()
add_subdirectory("${base}/audio-native" audio)
add_subdirectory("${base}/../src/Servers/Text/Native" text)
add_subdirectory("${base}/enet-native" enet)
if(EMSCRIPTEN)
    add_library(Electron2DWasmCompat STATIC wasm_sjlj.c)
    target_link_libraries(Electron2DFreeTypeCheck PRIVATE Electron2DWasmCompat)
    add_executable(Electron2DWasmJumpCheck EXCLUDE_FROM_ALL test_wasm_sjlj.c)
    target_link_libraries(Electron2DWasmJumpCheck PRIVATE Electron2DWasmCompat)
    target_link_options(Electron2DWasmJumpCheck PRIVATE -sWASM_ASYNC_COMPILATION=0)
endif()
