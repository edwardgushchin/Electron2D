cmake_minimum_required(VERSION 3.24)
project(Electron2DNative LANGUAGES C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/msvc-dependencies.cmake")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ELECTRON2D_UNIFIED_NATIVE ON)
set(ELECTRON2D_STATIC_NATIVE ON)
set(base "${CMAKE_CURRENT_LIST_DIR}/..")
add_subdirectory("${base}/enet-native" enet)
include("${CMAKE_CURRENT_LIST_DIR}/png.cmake")
add_subdirectory("${base}/../src/Servers/Text/Native" text)
foreach(component mesh render webp ogv ktx)
    add_subdirectory("${base}/${component}-native" ${component})
endforeach()
set(components Electron2DENet Electron2DTextBreak Electron2DMesh Electron2DRender
    Electron2DWebP Electron2DOGV Electron2DKTX)
if(NOT ANDROID AND NOT EMSCRIPTEN AND NOT CMAKE_SYSTEM_NAME MATCHES "^(iOS|tvOS)$")
    find_package(OpenSSL 4.1.0 EXACT REQUIRED)
    add_subdirectory("${base}/webrtc-native" webrtc)
    list(APPEND components Electron2DWebRTC)
endif()
if(EMSCRIPTEN)
    add_library(Electron2DWasmCompat STATIC "${CMAKE_CURRENT_LIST_DIR}/cross/wasm_sjlj.c")
    list(APPEND components Electron2DWasmCompat)
endif()
set(kind SHARED)
if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME MATCHES "^(iOS|tvOS)$")
    set(kind STATIC)
endif()
add_library(Electron2DNative ${kind} unified.c png.c)
target_compile_features(Electron2DNative PRIVATE c_std_11)
target_link_libraries(Electron2DNative PRIVATE png_static)
set_target_properties(Electron2DNative PROPERTIES OUTPUT_NAME Electron2D PREFIX lib
    C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" SKIP_BUILD_RPATH TRUE)
if(kind STREQUAL "STATIC")
    # A static archive must contain its closure; CMake does not merge linked archives.
    include("${CMAKE_CURRENT_LIST_DIR}/merge.cmake")
    e2d_merge_archives(Electron2DNative ${components})
else()
    include(FetchContent)
    FetchContent_GetProperties(SDLMemory SOURCE_DIR sdl_source)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    file(GLOB dispatch_inputs CONFIGURE_DEPENDS "${base}/render-native/*.c" "${base}/render-native/*.h")
    set(dispatch "${CMAKE_CURRENT_BINARY_DIR}/sdl-dispatch.c")
    add_custom_command(OUTPUT "${dispatch}"
        COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/sdl_dispatch.py"
            "${sdl_source}" "${base}/render-native" "${dispatch}"
        DEPENDS ${dispatch_inputs} "${CMAKE_CURRENT_LIST_DIR}/sdl_dispatch.py"
            "${sdl_source}/src/dynapi/SDL_dynapi_procs.h" "${base}/render-native/CMakeLists.txt"
        VERBATIM)
    target_sources(Electron2DNative PRIVATE "${dispatch}")
    target_include_directories(Electron2DNative PRIVATE "${sdl_source}/include")
    target_link_libraries(Electron2DNative PRIVATE ${CMAKE_DL_LIBS})
    target_link_libraries(Electron2DNative PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,${components}>")
    set_property(TARGET Electron2DNative PROPERTY LINK_LIBRARY_OVERRIDE "WHOLE_ARCHIVE,${components}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/exports.txt")
    set_property(TARGET Electron2DNative APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/exports.txt")
    file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/exports.txt" symbols)
    if(ANDROID)
        list(FILTER symbols EXCLUDE REGEX "^e2d_rtc_")
    endif()
    if(WIN32)
        string(JOIN "\n" exports ${symbols})
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.def" "LIBRARY libElectron2D\nEXPORTS\n${exports}\n")
        target_sources(Electron2DNative PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/exports.def")
        target_link_libraries(Electron2DNative PRIVATE bcrypt ws2_32)
    elseif(APPLE)
        list(TRANSFORM symbols PREPEND "_")
        string(JOIN "\n" exports ${symbols})
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.macos" "${exports}\n")
        target_link_options(Electron2DNative PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_BINARY_DIR}/exports.macos")
        set_target_properties(Electron2DNative PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
    else()
        set_target_properties(Electron2DNative PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
        string(JOIN ";\n" exports ${symbols})
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.map" "{ global:\n${exports};\nlocal: *;\n};\n")
        target_link_options(Electron2DNative PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/exports.map" "LINKER:--no-undefined" "LINKER:-Bsymbolic")
    endif()
endif()
