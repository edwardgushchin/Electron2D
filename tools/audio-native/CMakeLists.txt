cmake_minimum_required(VERSION 3.16)
project(Electron2DAudio C)
# Fetch only the matching headers; the SDL library comes from NuGet.
if(NOT ELECTRON2D_SDL_VERSION STREQUAL "3.4.16")
    message(FATAL_ERROR "Update the pinned SDL headers and archive hash with the native package version.")
endif()
if(NOT TARGET "${SDL3_LIBRARIES}" AND NOT EXISTS "${SDL3_LIBRARIES}")
    message(FATAL_ERROR "SDL3_LIBRARIES must name the selected RID's restored SDL3-CS native library.")
endif()
if(TARGET "${SDL3_LIBRARIES}")
    set(SDL3_INCLUDE_DIRS "")
else()
    include(FetchContent)
    FetchContent_Declare(SDL3Headers
    URL "https://www.libsdl.org/release/SDL3-3.4.16.tar.gz"
    URL_HASH SHA256=7322236cd12090c3eb40b9728be4d49c76f66ad17d04369584d4ecad5cf77c68
    SOURCE_SUBDIR "headers-only")
    FetchContent_MakeAvailable(SDL3Headers)
    set(SDL3_INCLUDE_DIRS "${sdl3headers_SOURCE_DIR}/include")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../src/Vendor/FAudio" faudio)
set(E2D_AUDIO_TARGET FAudio-shared)
if(TARGET FAudio-static)
    set(E2D_AUDIO_TARGET FAudio-static)
endif()
target_sources(${E2D_AUDIO_TARGET} PRIVATE output.c)
target_include_directories(${E2D_AUDIO_TARGET} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/Vendor/FAudio/src")
set_target_properties(${E2D_AUDIO_TARGET} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" SKIP_BUILD_RPATH TRUE)
if(ANDROID)
    set_property(TARGET ${E2D_AUDIO_TARGET} PROPERTY VERSION)
    set_property(TARGET ${E2D_AUDIO_TARGET} PROPERTY SOVERSION)
endif()
if(APPLE)
    set_target_properties(${E2D_AUDIO_TARGET} PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
endif()
