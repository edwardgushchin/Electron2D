cmake_minimum_required(VERSION 3.24)
project(Electron2DWebP LANGUAGES C)
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
foreach(option ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO WEBPMUX EXTRAS)
  set(WEBP_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(Electron2DWebPCodec
  URL https://storage.googleapis.com/downloads.webmproject.org/releases/webp/libwebp-1.6.0.tar.gz
  URL_HASH SHA256=e4ab7009bf0629fd11982d4c2aa83964cf244cffba7347ecd39019a9e38c4564)
if(NOT TARGET webp)
  FetchContent_MakeAvailable(Electron2DWebPCodec)
endif()
set(kind SHARED)
if(ELECTRON2D_STATIC_NATIVE)
  set(kind STATIC)
endif()
add_library(Electron2DWebP ${kind} bridge.c)
target_link_libraries(Electron2DWebP PRIVATE webpdemux webp)
target_compile_features(Electron2DWebP PRIVATE c_std_11)
target_include_directories(Electron2DWebP PRIVATE "${electron2dwebpcodec_SOURCE_DIR}/src")
set_target_properties(Electron2DWebP PROPERTIES C_VISIBILITY_PRESET hidden)
if(APPLE AND NOT ELECTRON2D_STATIC_NATIVE)
  set_target_properties(Electron2DWebP PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE SKIP_BUILD_RPATH TRUE)
elseif(WIN32)
  target_sources(Electron2DWebP PRIVATE exports.def)
endif()
add_executable(Electron2DWebPCheck EXCLUDE_FROM_ALL test_bridge.c bridge.c)
target_link_libraries(Electron2DWebPCheck PRIVATE webpdemux webp)
