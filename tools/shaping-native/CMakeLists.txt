cmake_minimum_required(VERSION 3.24)
project(Electron2DShaping LANGUAGES C CXX)
include(FetchContent)
include("${CMAKE_CURRENT_LIST_DIR}/prepare.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../font-native/verify-source.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../native/harfbuzz.cmake")
if(NOT EXISTS "${ELECTRON2D_MEMORY_LIBRARY}")
    message(FATAL_ERROR "Select the matching pure-C ledger with ELECTRON2D_MEMORY_LIBRARY")
endif()
set(revision ${ELECTRON2D_HARFBUZZ_REVISION})
if(FETCHCONTENT_SOURCE_DIR_SHAPING)
    verify_source(HarfBuzz "${revision}" "${FETCHCONTENT_SOURCE_DIR_SHAPING}")
endif()
FetchContent_Declare(Shaping GIT_REPOSITORY https://github.com/harfbuzz/harfbuzz.git
    GIT_TAG ${revision} GIT_SHALLOW FALSE SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(Shaping)
verify_source(HarfBuzz "${revision}" "${shaping_SOURCE_DIR}")
set(BUILD_SHARED_LIBS ON)
foreach(feature FREETYPE CORETEXT GRAPHITE2 GLIB ICU GOBJECT CAIRO UNISCRIBE GDI DIRECTWRITE)
    set(HB_HAVE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature SUBSET UTILS RASTER VECTOR GPU)
    set(HB_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${shaping_SOURCE_DIR}" harfbuzz)
target_compile_definitions(harfbuzz PRIVATE HB_NO_FALLBACK_SHAPE)
if(NOT ELECTRON2D_SHAPING_REFERENCE_PROFILE)
    e2d_prepare_shaping(harfbuzz "${shaping_SOURCE_DIR}" e2d_memory_failure_epoch)
    set(adapter allocator.cc)
    if(ELECTRON2D_SHAPING_FAULT_PROFILE)
        set(adapter fault-allocator.cc)
    endif()
    target_sources(harfbuzz PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${adapter}")
    target_compile_definitions(harfbuzz PRIVATE E2D_MEMORY_IMPORT
        hb_malloc_impl=e2d_shaping_malloc hb_calloc_impl=e2d_shaping_calloc
        hb_realloc_impl=e2d_shaping_realloc hb_free_impl=e2d_shaping_free)
    target_link_libraries(harfbuzz "${ELECTRON2D_MEMORY_LIBRARY}")
endif()
if(NOT WIN32)
    target_compile_definitions(harfbuzz PRIVATE "HB_EXTERN=extern __attribute__((visibility(\"default\")))")
endif()
set_target_properties(harfbuzz PROPERTIES OUTPUT_NAME HarfBuzzSharp PREFIX lib
    CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden SKIP_BUILD_RPATH TRUE
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set_property(TARGET harfbuzz PROPERTY VERSION)
set_property(TARGET harfbuzz PROPERTY SOVERSION)
if(APPLE)
    set_target_properties(harfbuzz PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
endif()
if(UNIX AND NOT APPLE)
    add_executable(Electron2DShapingCheck EXCLUDE_FROM_ALL check.cc)
    target_compile_features(Electron2DShapingCheck PRIVATE cxx_std_17)
    target_include_directories(Electron2DShapingCheck PRIVATE "${shaping_SOURCE_DIR}/src" "${CMAKE_CURRENT_BINARY_DIR}/harfbuzz/src")
    target_link_libraries(Electron2DShapingCheck PRIVATE "${ELECTRON2D_MEMORY_LIBRARY}" ${CMAKE_DL_LIBS})
endif()
