cmake_minimum_required(VERSION 3.20)
project(Electron2DWebRTC LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_DEPS_LIBS OFF CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(USE_NETTLE OFF CACHE BOOL "" FORCE)
find_package(OpenSSL 4.1.0 EXACT REQUIRED)
add_subdirectory(../../src/Vendor/libdatachannel dependency EXCLUDE_FROM_ALL)
set(kind SHARED)
if(ELECTRON2D_STATIC_NATIVE)
  set(kind STATIC)
endif()
add_library(Electron2DWebRTC ${kind} bridge.cpp)
target_link_libraries(Electron2DWebRTC PRIVATE datachannel OpenSSL::Crypto)
if(NOT ELECTRON2D_UNIFIED_NATIVE)
  find_package(Threads REQUIRED)
  add_library(Electron2DRTCMemory STATIC ../enet-native/memory.c)
  target_compile_features(Electron2DRTCMemory PRIVATE c_std_11)
  target_link_libraries(Electron2DRTCMemory PRIVATE Threads::Threads)
  target_link_libraries(Electron2DWebRTC PRIVATE Electron2DRTCMemory)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(Electron2DRTCMemoryCheck EXCLUDE_FROM_ALL test_memory.cpp fault.c)
  target_compile_features(Electron2DRTCMemoryCheck PRIVATE c_std_11 cxx_std_17)
  target_compile_options(Electron2DRTCMemoryCheck PRIVATE -UNDEBUG)
  target_link_libraries(Electron2DRTCMemoryCheck PRIVATE datachannel OpenSSL::Crypto)
  target_link_options(Electron2DRTCMemoryCheck PRIVATE "LINKER:--export-dynamic")
  if(NOT ELECTRON2D_UNIFIED_NATIVE)
    file(WRITE "${CMAKE_BINARY_DIR}/rtc-consumer.map" "{ global: e2d_rtc_*; e2d_memory_*; local: *; };")
    foreach(profile IN ITEMS consumer close-failure)
      if(profile STREQUAL "consumer")
        set(check_target Electron2DRTCConsumerCheck)
        set(check_source bridge.cpp)
      else()
        set(check_target Electron2DRTCCloseFailureCheck)
        set(check_source test_close.cpp)
      endif()
      add_library(${check_target} SHARED EXCLUDE_FROM_ALL ${check_source} ../enet-native/memory.c)
      target_compile_features(${check_target} PRIVATE c_std_11 cxx_std_17)
      target_link_libraries(${check_target} PRIVATE datachannel OpenSSL::Crypto Threads::Threads)
      set_target_properties(${check_target} PROPERTIES OUTPUT_NAME Electron2D
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${profile}" BUILD_WITH_INSTALL_RPATH YES
        INSTALL_RPATH "$ORIGIN" CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden)
      target_link_options(${check_target} PRIVATE "LINKER:--version-script=${CMAKE_BINARY_DIR}/rtc-consumer.map" "LINKER:-z,defs" "LINKER:-Bsymbolic")
    endforeach()
  endif()
endif()
set_target_properties(Electron2DWebRTC PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE AND NOT ELECTRON2D_STATIC_NATIVE)
  set_target_properties(Electron2DWebRTC PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR YES SKIP_BUILD_RPATH YES)
  target_link_options(Electron2DWebRTC PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.macos")
elseif(UNIX AND NOT ELECTRON2D_STATIC_NATIVE)
  set_property(TARGET Electron2DWebRTC APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
  target_link_options(Electron2DWebRTC PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
  set_target_properties(Electron2DWebRTC PROPERTIES BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
endif()
