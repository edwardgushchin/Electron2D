cmake_minimum_required(VERSION 3.24)
project(Electron2DSDLMemory LANGUAGES C)
include(FetchContent)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
if(NOT TARGET Electron2DENet)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../enet-native" ledger)
endif()
FetchContent_Declare(SDLMemory
    URL https://www.libsdl.org/release/SDL3-3.4.18.tar.gz
    URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3
    SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(SDLMemory)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL_VENDOR_INFO "Electron2D memory 1" CACHE STRING "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
if(ELECTRON2D_STATIC_NATIVE)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${sdlmemory_SOURCE_DIR}" sdl)
set(target SDL3-shared)
if(ELECTRON2D_STATIC_NATIVE)
    set(target SDL3-static)
endif()
get_target_property(sources ${target} SOURCES)
foreach(name stdlib/SDL_malloc stdlib/SDL_getenv SDL_properties io/SDL_iostream)
    set(original "${sdlmemory_SOURCE_DIR}/src/${name}.c")
    get_filename_component(name "${name}" NAME)
    set(generated "${CMAKE_CURRENT_BINARY_DIR}/${name}.c")
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_memory.py"
        "${original}" "${generated}" COMMAND_ERROR_IS_FATAL ANY)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS prepare_memory.py "${original}")
    list(FIND sources "${original}" index)
    if(index EQUAL -1)
        message(FATAL_ERROR "The pinned SDL target no longer compiles ${original}")
    endif()
    list(REMOVE_ITEM sources "${original}")
    list(APPEND sources "${generated}")
endforeach()
set_property(TARGET ${target} PROPERTY SOURCES "${sources}")
target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../enet-native" "${sdlmemory_SOURCE_DIR}/src/stdlib" "${sdlmemory_SOURCE_DIR}/src/io")
target_compile_definitions(${target} PRIVATE E2D_MEMORY_IMPORT SDL_TRACK_ALLOCATION_COUNT)
target_link_libraries(${target} PRIVATE "$<TARGET_LINKER_FILE:Electron2DENet>")
add_dependencies(${target} Electron2DENet)
set_target_properties(${target} PROPERTIES SKIP_BUILD_RPATH TRUE)
add_executable(Electron2DSDLMemoryCheck EXCLUDE_FROM_ALL check.c)
target_compile_definitions(Electron2DSDLMemoryCheck PRIVATE E2D_MEMORY_IMPORT)
target_link_libraries(Electron2DSDLMemoryCheck PRIVATE ${target} Electron2DENet ${CMAKE_DL_LIBS})
add_library(Electron2DSDLLifetime SHARED EXCLUDE_FROM_ALL lifetime.c)
target_compile_definitions(Electron2DSDLLifetime PRIVATE E2D_MEMORY_IMPORT)
target_link_libraries(Electron2DSDLLifetime PRIVATE ${target} Electron2DENet)
set_target_properties(Electron2DSDLLifetime PROPERTIES C_VISIBILITY_PRESET hidden SKIP_BUILD_RPATH TRUE)
if(UNIX AND NOT APPLE)
    add_library(Electron2DSDLFaultCore SHARED EXCLUDE_FROM_ALL ../runtime-native/compression/fault.c)
    target_compile_features(Electron2DSDLFaultCore PRIVATE c_std_11)
    target_link_libraries(Electron2DSDLFaultCore PRIVATE Threads::Threads)
    set_target_properties(Electron2DSDLFaultCore PROPERTIES C_VISIBILITY_PRESET hidden
        OUTPUT_NAME Electron2DENet LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fault")
endif()
