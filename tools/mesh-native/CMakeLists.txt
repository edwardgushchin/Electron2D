cmake_minimum_required(VERSION 3.24)
project(Electron2DMesh LANGUAGES CXX)
set(kind SHARED)
if(ELECTRON2D_STATIC_NATIVE)
  set(kind STATIC)
endif()
set(vendor "${CMAKE_CURRENT_SOURCE_DIR}/../../src/Vendor/meshoptimizer")
add_library(Electron2DMesh ${kind} bridge.cpp "${vendor}/allocator.cpp" "${vendor}/simplifier.cpp")
target_include_directories(Electron2DMesh PRIVATE "${vendor}")
target_compile_features(Electron2DMesh PRIVATE cxx_std_11)
set_target_properties(Electron2DMesh PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE ON)
if(APPLE AND NOT ELECTRON2D_STATIC_NATIVE)
  set_target_properties(Electron2DMesh PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE SKIP_BUILD_RPATH TRUE)
  target_link_options(Electron2DMesh PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.macos")
elseif(WIN32)
  target_sources(Electron2DMesh PRIVATE exports.def)
elseif(UNIX AND NOT ELECTRON2D_STATIC_NATIVE)
  set_property(TARGET Electron2DMesh APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
  target_link_options(Electron2DMesh PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
add_executable(Electron2DMeshOracle EXCLUDE_FROM_ALL oracle.cpp "${vendor}/allocator.cpp" "${vendor}/simplifier.cpp")
target_include_directories(Electron2DMeshOracle PRIVATE "${vendor}")
target_compile_features(Electron2DMeshOracle PRIVATE cxx_std_11)
