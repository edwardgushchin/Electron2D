cmake_minimum_required(VERSION 3.24)
project(Electron2DKTX LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(KTX_GIT_VERSION_FULL v5.0.0-rc2 CACHE STRING "" FORCE)
foreach(option DOC JNI PY JS TESTS TOOLS TOOLS_CTS)
  set(KTX_FEATURE_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LIBKTX_VERSION_FULL OFF CACHE BOOL "" FORCE)
set(LIBKTX_VERSION_READ_ONLY ON CACHE BOOL "" FORCE)
set(LIBKTX_FEATURE_ETC_UNPACK OFF CACHE BOOL "" FORCE)
set(LIBKTX_FEATURE_GL_UPLOAD OFF CACHE BOOL "" FORCE)
set(LIBKTX_FEATURE_VK_UPLOAD OFF CACHE BOOL "" FORCE)
set(BASISU_TOOL OFF CACHE BOOL "" FORCE)
set(BASISU_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BASISU_SSE OFF CACHE BOOL "" FORCE)
set(BASISU_OPENCL OFF CACHE BOOL "" FORCE)
set(KTX_FEATURE_LOADTEST_APPS OFF CACHE STRING "" FORCE)
set(ASTCENC_ISA_NONE ON CACHE BOOL "" FORCE)
FetchContent_Declare(Electron2DKTXCodec
  URL https://codeload.github.com/KhronosGroup/KTX-Software/tar.gz/refs/tags/v5.0.0-rc2
  URL_HASH SHA256=bce6794bca57d86de2f00683ba8eaa6b3e8875d5abe45ac1136e206eb68aa476)
FetchContent_MakeAvailable(Electron2DKTXCodec)
include("${CMAKE_CURRENT_SOURCE_DIR}/decoder-guards.cmake")
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
  # The upstream C vtables intentionally erase the derived object pointer type.
  target_compile_options(ktx_read PRIVATE -fno-sanitize=function)
endif()
# Namespace the unchanged amalgamation, including its internal macro undefinitions.
set(zstd_source "${electron2dktxcodec_SOURCE_DIR}/external/basis_universal/zstd/zstd.c")
file(GLOB zstd_headers "${electron2dktxcodec_SOURCE_DIR}/external/basis_universal/zstd/*.h")
foreach(source "${zstd_source}" ${zstd_headers})
  get_filename_component(name "${source}" NAME)
  file(READ "${source}" zstd)
  string(REGEX REPLACE "(^|[^A-Za-z0-9_])(ZSTD_|ZSTDMT_|XXH_|POOL_|FSE_|HUF_|HIST_|ZDICT_|COVER_|FASTCOVER_|ERR_)" "\\1e2d_ktx_private_\\2" zstd "${zstd}")
  set(output "${CMAKE_CURRENT_BINARY_DIR}/${name}")
  set(previous "")
  if(EXISTS "${output}")
    file(READ "${output}" previous)
  endif()
  if(NOT previous STREQUAL zstd)
    file(WRITE "${output}" "${zstd}")
  endif()
endforeach()
set_source_files_properties("${zstd_source}" TARGET_DIRECTORY basisu_encoder PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(basisu_encoder PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/zstd.c")
foreach(target ktx_read basisu_encoder)
  if(MSVC)
    target_compile_options(${target} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/zstd-namespace.h")
  else()
    target_compile_options(${target} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/zstd-namespace.h")
  endif()
endforeach()
set(kind SHARED)
if(ELECTRON2D_STATIC_NATIVE)
  set(kind STATIC)
endif()
add_library(Electron2DKTX ${kind} bridge.cpp)
target_compile_features(Electron2DKTX PRIVATE cxx_std_17)
target_link_libraries(Electron2DKTX PRIVATE ktx_read)
target_compile_definitions(Electron2DKTX PRIVATE E2D_KTX_BUILD)
if(ELECTRON2D_STATIC_NATIVE)
  target_compile_definitions(Electron2DKTX PUBLIC E2D_KTX_STATIC)
  # Static link dependencies are order-only; refresh the merged closure when its codec changes.
  set(stamp "${CMAKE_CURRENT_BINARY_DIR}/ktx-closure.stamp")
  add_custom_command(OUTPUT "${stamp}"
    COMMAND ${CMAKE_COMMAND} -E touch "${stamp}"
    DEPENDS ktx_read "${CMAKE_CURRENT_SOURCE_DIR}/merge.cmake")
  target_sources(Electron2DKTX PRIVATE "${stamp}")
  set_property(SOURCE bridge.cpp APPEND PROPERTY OBJECT_DEPENDS "${stamp}")
  add_custom_command(TARGET Electron2DKTX POST_BUILD
    COMMAND ${CMAKE_COMMAND} "-DOUTPUT=$<TARGET_FILE:Electron2DKTX>" "-DCODEC=$<TARGET_FILE:ktx_read>"
      "-DAR=${CMAKE_AR}" "-DAPPLE_HOST=${APPLE}" -P "${CMAKE_CURRENT_SOURCE_DIR}/merge.cmake")
elseif(APPLE)
  set_target_properties(Electron2DKTX PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE SKIP_BUILD_RPATH TRUE)
elseif(WIN32)
  target_sources(Electron2DKTX PRIVATE exports.def)
elseif(UNIX)
  target_link_options(Electron2DKTX PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:-z,defs")
endif()
add_executable(Electron2DKTXCheck EXCLUDE_FROM_ALL test_bridge.cpp)
target_compile_features(Electron2DKTXCheck PRIVATE cxx_std_17)
target_link_libraries(Electron2DKTXCheck PRIVATE Electron2DKTX)
