file(READ "${electron2detc_SOURCE_DIR}/detex.h" detex_header)
set(inline_preimage "#define DETEX_INLINE_ONLY __attribute__((always_inline)) inline")
string(FIND "${detex_header}" "${inline_preimage}" found)
if(found LESS 0)
  message(FATAL_ERROR "Pinned detex compiler adapter no longer matches")
endif()
string(REPLACE "${inline_preimage}" "#if defined(_MSC_VER)\n#define DETEX_INLINE_ONLY __forceinline\n#else\n${inline_preimage}\n#endif" detex_header "${detex_header}")
set(header_output "${CMAKE_CURRENT_BINARY_DIR}/detex.h")
set(previous "")
if(EXISTS "${header_output}")
  file(READ "${header_output}" previous)
endif()

  file(WRITE "${header_output}" "${detex_header}")
endif()

string(FIND "${eac}" "base_codeword << 3" signed_shift)
if(signed_shift LESS 0)
  message(FATAL_ERROR "Pinned EAC arithmetic guard no longer matches")
endif()
string(REPLACE "base_codeword << 3" "base_codeword * 8" eac "${eac}")
set(eac_output "${CMAKE_CURRENT_BINARY_DIR}/decompress-eac.c")
set(previous "")
if(EXISTS "${eac_output}")
  file(READ "${eac_output}" previous)

  file(WRITE "${eac_output}" "${eac}")
endif()

  string(FIND "${etc}" "${preimage}" found)
  if(found LESS 0)
    message(FATAL_ERROR "Pinned ETC arithmetic guard no longer matches: ${preimage}")
  endif()
endforeach()
string(REPLACE "0x10000 << i" "0x10000u << i" etc "${etc}")
string(REPLACE "((x & 3) - 4) << 3" "((x & 3) - 4) * 8" etc "${etc}")
set(etc_output "${CMAKE_CURRENT_BINARY_DIR}/decompress-etc.c")

  file(READ "${etc_output}" previous)

# Namespace the unchanged amalgamation, including its internal macro undefinitions.
set(zstd_source "${electron2dktxcodec_SOURCE_DIR}/external/basis_universal/zstd/zstd.c")
file(GLOB zstd_headers "${electron2dktxcodec_SOURCE_DIR}/external/basis_universal/zstd/*.h")
foreach(source "${zstd_source}" ${zstd_headers})
  get_filename_component(name "${source}" NAME)
  file(READ "${source}" zstd)
  string(REGEX REPLACE "(^|[^A-Za-z0-9_])(ZSTD_|ZSTDMT_|XXH_|POOL_|FSE_|HUF_|HIST_|ZDICT_|COVER_|FASTCOVER_|ERR_)" "\\1e2d_ktx_private_\\2" zstd "${zstd}")
  set(output "${CMAKE_CURRENT_BINARY_DIR}/${name}")
  set(previous "")
  if(EXISTS "${output}")
    file(READ "${output}" previous)
  endif()
  if(NOT previous STREQUAL zstd)
    file(WRITE "${output}" "${zstd}")
  endif()
endforeach()
set_source_files_properties("${zstd_source}" TARGET_DIRECTORY basisu_encoder PROPERTIES HEADER_FILE_ONLY TRUE)
