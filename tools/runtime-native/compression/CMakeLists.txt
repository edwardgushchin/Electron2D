cmake_minimum_required(VERSION 3.20)
project(Electron2DCompressionRuntime C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
    message(FATAL_ERROR "This source-runtime component is currently qualified only for Linux x64.")
endif()
if(NOT EXISTS "${E2D_RUNTIME_ARCHIVE}" OR NOT EXISTS "${E2D_MEMORY_LIBRARY}")
    message(FATAL_ERROR "Supply the pinned runtime archive and existing production memory bundle.")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/prepare.py" preparer_hash)
set(source "${CMAKE_CURRENT_BINARY_DIR}/source-${preparer_hash}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/prepare.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/prepare.py" "${E2D_RUNTIME_ARCHIVE}" "${source}" COMMAND_ERROR_IS_FATAL ANY)
set(runtime "${source}/runtime-10.0.1/src/native")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CLR_CMAKE_TARGET_LINUX TRUE)
set(WITH_NATIVE_INSTRUCTIONS OFF CACHE BOOL "" FORCE)
include("${runtime}/external/zlib-ng.cmake")
include("${runtime}/external/brotli.cmake")
target_include_directories(brotlicommon PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_include_directories(brotlienc PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_include_directories(brotlidec PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(brotlienc PRIVATE BROTLI_ENCODER_CLEANUP_ON_OOM)
configure_file("${runtime}/libs/Common/pal_config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/pal_config.h")
add_library(System.IO.Compression.Native SHARED
    "${runtime}/libs/System.IO.Compression.Native/pal_zlib.c"
    "${runtime}/libs/System.IO.Compression.Native/entrypoints.c"
    abi.c)
target_include_directories(System.IO.Compression.Native PRIVATE "${runtime}/libs/Common" "${runtime}" "${runtime}/external/brotli/c/include" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(System.IO.Compression.Native PRIVATE zlibstatic brotlienc brotlidec brotlicommon "${E2D_MEMORY_LIBRARY}")
target_compile_definitions(System.IO.Compression.Native PRIVATE TARGET_UNIX BROTLI_SHARED_COMPILATION)
file(STRINGS "${runtime}/libs/System.IO.Compression.Native/System.IO.Compression.Native_unixexports.src" exports REGEX "^[A-Za-z]")
list(APPEND exports e2d_compression_memory_abi)
string(JOIN ";\n" exports ${exports})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.map" "{ global:\n${exports};\nlocal: *; };\n")
target_link_options(System.IO.Compression.Native PRIVATE "-Wl,--no-undefined" "-Wl,-Bsymbolic-functions" "-Wl,--version-script=${CMAKE_CURRENT_BINARY_DIR}/exports.map")
set_target_properties(System.IO.Compression.Native PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/runtimes/linux-x64/native")

option(E2D_COMPRESSION_TESTS "Build the isolated allocator-failure consumer" OFF)
if(E2D_COMPRESSION_TESTS)
    add_library(CompressionFaultCore SHARED fault.c)
    target_link_libraries(CompressionFaultCore PRIVATE pthread)
    set_target_properties(CompressionFaultCore PROPERTIES OUTPUT_NAME Electron2DENet LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fault")
    add_executable(Electron2DCompressionMemoryCheck check.c)
    target_compile_definitions(Electron2DCompressionMemoryCheck PRIVATE TARGET_UNIX)
    target_include_directories(Electron2DCompressionMemoryCheck PRIVATE "${runtime}/libs/System.IO.Compression.Native" "${runtime}/libs/Common" "${runtime}/external/brotli/c/include")
    target_link_libraries(Electron2DCompressionMemoryCheck PRIVATE System.IO.Compression.Native CompressionFaultCore)
endif()
