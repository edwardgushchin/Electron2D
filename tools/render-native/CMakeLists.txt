cmake_minimum_required(VERSION 3.24)
project(Electron2DRender LANGUAGES C)
if(NOT SDL3_LIBRARIES)
    message(FATAL_ERROR "The selected target's pinned SDL core library is required")
endif()
include(FetchContent)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
FetchContent_Declare(SDLMemory
    URL https://www.libsdl.org/release/SDL3-3.4.18.tar.gz
    URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3
    SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(SDLMemory)
set(layout "${CMAKE_CURRENT_BINARY_DIR}/memory-layout.h")
add_custom_command(OUTPUT "${layout}"
    COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/generate_layout.py" "${sdlmemory_SOURCE_DIR}" "${layout}"
    DEPENDS "${sdlmemory_SOURCE_DIR}/src/gpu/vulkan/SDL_gpu_vulkan_vkfuncs.h" generate_layout.py "${sdlmemory_SOURCE_DIR}/src/gpu/SDL_sysgpu.h" "${sdlmemory_SOURCE_DIR}/src/gpu/vulkan/SDL_gpu_vulkan.c")
set(kind SHARED)
if(ELECTRON2D_STATIC_NATIVE)
    set(kind STATIC)
endif()
add_library(Electron2DRender ${kind} bridge.c sampler.c limits.c filtering.c storage.c empty.c raster.c "${layout}")
target_compile_features(Electron2DRender PRIVATE c_std_11)
target_include_directories(Electron2DRender PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${sdlmemory_SOURCE_DIR}/include" "${sdlmemory_SOURCE_DIR}/src/video/khronos")
target_link_libraries(Electron2DRender PRIVATE ${SDL3_LIBRARIES})
set_target_properties(Electron2DRender PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON SKIP_BUILD_RPATH TRUE)
if(APPLE AND NOT ELECTRON2D_STATIC_NATIVE)
    set_target_properties(Electron2DRender PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE SKIP_BUILD_RPATH TRUE)
    target_link_options(Electron2DRender PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.macos")
elseif(WIN32)
    target_sources(Electron2DRender PRIVATE exports.def)
elseif(UNIX AND NOT ELECTRON2D_STATIC_NATIVE)
    set_property(TARGET Electron2DRender APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
    target_link_options(Electron2DRender PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
add_executable(Electron2DRenderCheck EXCLUDE_FROM_ALL check.c)
target_compile_features(Electron2DRenderCheck PRIVATE c_std_11)
target_include_directories(Electron2DRenderCheck PRIVATE "${sdlmemory_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}" "${sdlmemory_SOURCE_DIR}/src/video/khronos")
target_link_libraries(Electron2DRenderCheck PRIVATE Electron2DRender ${SDL3_LIBRARIES})
