cmake_minimum_required(VERSION 3.20)
project(Electron2DTextBreak LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ELECTRON2D_ICU_BUILD_TOOL "Build the pinned host data packaging tool" OFF)
set(ELECTRON2D_TEXT_TARGET_ARCH "" CACHE STRING "Expected target processor: x64 or arm64")

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" E2D_PROCESSOR)
if(ELECTRON2D_TEXT_TARGET_ARCH STREQUAL "x64" AND NOT E2D_PROCESSOR MATCHES "^(x86_64|amd64)$")
    message(FATAL_ERROR "The selected C++ toolchain does not target x64; use the matching native toolchain.")
elseif(ELECTRON2D_TEXT_TARGET_ARCH STREQUAL "arm64" AND NOT E2D_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "The selected C++ toolchain does not target arm64; use the matching native toolchain.")
endif()

get_filename_component(ICU_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../Vendor/ICU" ABSOLUTE)
file(READ "${ICU_ROOT}/manifest.json" ICU_MANIFEST)
function(read_sources key destination)
    string(JSON count LENGTH "${ICU_MANIFEST}" ${key})
    math(EXPR last "${count} - 1")
    set(values)
    foreach(index RANGE 0 ${last})
        string(JSON name GET "${ICU_MANIFEST}" ${key} ${index})
        list(APPEND values "${ICU_ROOT}/${name}")
    endforeach()
    set(${destination} "${values}" PARENT_SCOPE)
endfunction()
read_sources(runtimeSources ICU_RUNTIME_SOURCES)

set(ICU_DEFINITIONS
    U_STATIC_IMPLEMENTATION U_COMMON_IMPLEMENTATION
    UCONFIG_NO_COLLATION=1 UCONFIG_NO_CONVERSION=1 UCONFIG_NO_FORMATTING=1
    UCONFIG_NO_SERVICE=1 UCONFIG_NO_IDNA=1 UCONFIG_NO_FILE_IO=1
    UCONFIG_NO_TRANSLITERATION=1 UCONFIG_NO_REGULAR_EXPRESSIONS=1
    U_ENABLE_DYLOAD=0 U_HAVE_LIB_SUFFIX=1 U_LIB_SUFFIX_C_NAME=_electron2d)
add_library(Electron2DICUCommon OBJECT ${ICU_RUNTIME_SOURCES})
target_include_directories(Electron2DICUCommon PRIVATE "${ICU_ROOT}/common" "${ICU_ROOT}/stubdata")
target_compile_definitions(Electron2DICUCommon PRIVATE ${ICU_DEFINITIONS})
set_target_properties(Electron2DICUCommon PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

set(E2D_TEXT_KIND SHARED)
if(ELECTRON2D_STATIC_NATIVE)
    set(E2D_TEXT_KIND STATIC)
endif()
add_library(Electron2DTextBreak ${E2D_TEXT_KIND} TextBreak.cpp $<TARGET_OBJECTS:Electron2DICUCommon>)
target_include_directories(Electron2DTextBreak PRIVATE "${ICU_ROOT}/common")
target_compile_definitions(Electron2DTextBreak PRIVATE ${ICU_DEFINITIONS})
set_target_properties(Electron2DTextBreak PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE)
    set_target_properties(Electron2DTextBreak PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE SKIP_BUILD_RPATH TRUE)
endif()
if(MSVC)
    file(READ "${CMAKE_CURRENT_LIST_DIR}/exports-macos.txt" E2D_EXPORTS)
    string(REGEX REPLACE "(^|\n)_" "\\1" E2D_EXPORTS "${E2D_EXPORTS}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.def" "EXPORTS\n${E2D_EXPORTS}")
    target_sources(Electron2DTextBreak PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/exports.def")
    target_compile_options(Electron2DICUCommon PRIVATE /O2 /utf-8)
    target_compile_options(Electron2DTextBreak PRIVATE /O2 /utf-8)
    target_link_options(Electron2DTextBreak PRIVATE /OPT:REF /OPT:ICF)
else()
    target_compile_options(Electron2DICUCommon PRIVATE -O2 -ffunction-sections -fdata-sections)
    target_compile_options(Electron2DTextBreak PRIVATE -O2 -ffunction-sections -fdata-sections)
    if(ELECTRON2D_STATIC_NATIVE)
        # Archive symbols are resolved by the target application linker.
    elseif(APPLE)
        target_link_options(Electron2DTextBreak PRIVATE -Wl,-dead_strip "-Wl,-exported_symbols_list,${CMAKE_CURRENT_LIST_DIR}/exports-macos.txt")
    else()
        target_link_options(Electron2DTextBreak PRIVATE -Wl,--gc-sections -Wl,--no-undefined "-Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/exports.map")
    endif()
endif()

if(ELECTRON2D_ICU_BUILD_TOOL)
    if(CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "ICU data packaging tools must be built for the host, separately from the target runtime library.")
    endif()
    read_sources(hostToolSources ICU_TOOL_SOURCES)
    add_executable(e2d_icupkg ${ICU_TOOL_SOURCES} $<TARGET_OBJECTS:Electron2DICUCommon>)
    target_include_directories(e2d_icupkg PRIVATE "${ICU_ROOT}/common" "${ICU_ROOT}/i18n" "${ICU_ROOT}/tools/toolutil")
    target_compile_definitions(e2d_icupkg PRIVATE ${ICU_DEFINITIONS} U_TOOLUTIL_IMPLEMENTATION)
    # This tool-only parser reads converter headers without enabling the converter runtime.
    if(MSVC)
        set_source_files_properties("${ICU_ROOT}/tools/toolutil/pkgitems.cpp" PROPERTIES COMPILE_OPTIONS "/UUCONFIG_NO_CONVERSION;/DUCONFIG_NO_CONVERSION=0")
        target_compile_options(e2d_icupkg PRIVATE /O2 /utf-8)
        target_link_options(e2d_icupkg PRIVATE /OPT:REF /OPT:ICF)
    else()
        set_source_files_properties("${ICU_ROOT}/tools/toolutil/pkgitems.cpp" PROPERTIES COMPILE_OPTIONS "-UUCONFIG_NO_CONVERSION;-DUCONFIG_NO_CONVERSION=0")
        target_compile_options(e2d_icupkg PRIVATE -O2 -ffunction-sections -fdata-sections)
        if(APPLE)
            target_link_options(e2d_icupkg PRIVATE -Wl,-dead_strip)
        else()
            target_link_options(e2d_icupkg PRIVATE -Wl,--gc-sections)
        endif()
    endif()
endif()
